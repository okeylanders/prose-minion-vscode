import { MessageType } from '@messages';
import { harness } from './workshopPinnedContextHandlerHost';
it('PROBE: real host newer list request can answer before an older pin commits', async () => {
 const h = harness({catalog:[{group:'characters',path:'Characters/alice.md'}]});
 h.write('Characters/alice.md','Alice is seven.');
 const initial = await h.send(MessageType.WORKSHOP_REQUEST_PINNED_CONTEXT,{});
 const open = h.intake.openCatalog.bind(h.intake);
 let release!: () => void;
 let entered!: () => void;
 const reached = new Promise<void>(resolve => {entered = resolve;});
 const held = new Promise<void>(resolve => {release = resolve;});
 jest.spyOn(h.intake,'openCatalog').mockImplementationOnce(async () => {entered(); await held; return open();});
 const pin = h.send(MessageType.WORKSHOP_PIN_CONTEXT_RESOURCES,{revision:initial.revision,items:[{group:'characters',path:'Characters/alice.md'}]});
 await reached;
 const newer = await h.send(MessageType.WORKSHOP_REQUEST_PINNED_CONTEXT,{});
 expect(newer.entries).toEqual([]);
 release();
 const completed = await pin;
 expect(completed.ok).toBe(true);
 expect(completed.entries).toHaveLength(1);
 expect(h.listOnDisk()).toEqual(['Characters/alice.md']);
 expect(h.posted.map(payload=>payload.requestId)).toEqual(['req-1','req-3','req-2']);
});

it('PROBE: real handler can attach B to a small old A snapshot despite template total exceeding budget', async () => {
 const h = harness();
 h.write('Characters/alice.md', 'Alice '.repeat(70000));
 h.write('Characters/the-hatter.md', 'Hatter '.repeat(40000));
 h.pin('Characters/alice.md','Characters/the-hatter.md');
 h.attachFile('Characters/alice.md',false); // Existing room keeps seven words, not current source's 70k.
 const result = await h.send(MessageType.WORKSHOP_ATTACH_PINNED_CONTEXT,{});
 expect(result.ok).toBe(true);
 expect(h.session.getContextAttachments().map(file=>file.relativePath)).toEqual(['Characters/alice.md','Characters/the-hatter.md']);
 expect(h.session.getContextAttachments().reduce((sum,file)=>sum+file.words,0)).toBe(40007);
});
