/** @jest-environment jsdom */

/**
 * PR #236 re-review at 9bfce925. Copy beside the conversation-modal tests.
 * A delayed host snapshot is deliberate: the setting is already Off, but
 * the webview still displays Ask first. These are corrected-contract tests,
 * so the untouched-row and omitted-hook cases fail before the correction and pass at 300a0e61.
 * No production components, hooks, router or settings service are mocked.
 */
import * as React from 'react';
import { TextEncoder, TextDecoder } from 'util';
import { act, fireEvent, render, renderHook, screen } from '@testing-library/react';
import { WorkshopConversationBehaviorModal } from '@components/workshop/WorkshopConversationBehaviorModal';
import { useWorkshopRoom } from '@hooks/domain/workshop/useWorkshopRoom';
import { useVSCodeApi } from '@hooks/useVSCodeApi';
import { createMockVSCode } from '@/__tests__/mocks/vscode';
import type { SettingsStore } from '@/platform';
import {
  DEFAULT_WORKSHOP_CONVERSATION_BEHAVIOR,
  DEFAULT_WORKSHOP_WRITER_PROFILE,
  DEFAULT_WORKSHOP_WEB_RESEARCH_SETTINGS,
  MessageType,
  type WorkshopSessionStateMessage
} from '@messages';
import { pinnedContextPanelPropsStub } from './settings/pinnedContextPanelPropsStub';

jest.mock('@hooks/useVSCodeApi');
Object.assign(globalThis, { TextEncoder, TextDecoder });

async function setup() {
  const { createWorkshopRouteTestHarness, message } = await import('@/__tests__/application/handlers/domain/workshop/WorkshopRouteTestHarness');
  const host = createWorkshopRouteTestHarness();
  const modes: Record<string, 'ask' | 'off'> = {
    'workshop.fileEdits': 'ask',
    'workshop.projectMemory': 'ask'
  };
  const originalGet = host.settings.get.bind(host.settings);
  (host.settings as { get: SettingsStore['get'] }).get = ((section: string, key: string, fallback?: unknown) =>
    key in modes ? modes[key] : originalGet(section, key, fallback)) as SettingsStore['get'];
  (host.settings.update as jest.Mock).mockImplementation(async (_section: string, key: string, value: unknown) => {
    if (key in modes) modes[key] = value as 'ask' | 'off';
  });
  await host.pin();
  await host.router.route(message(MessageType.WORKSHOP_SEND_MESSAGE, { text: 'Open the room.' }) as never);
  const vscode = createMockVSCode();
  (useVSCodeApi as jest.Mock).mockReturnValue(vscode);
  const { result } = renderHook(() => useWorkshopRoom());
  const snapshot = () => host.posted(MessageType.WORKSHOP_SESSION_STATE).at(-1) as WorkshopSessionStateMessage;
  act(() => result.current.handleSessionState(snapshot()));
  expect(result.current.fileEditsMode).toBe('ask');
  expect(result.current.projectMemoryMode).toBe('ask');
  const props: React.ComponentProps<typeof WorkshopConversationBehaviorModal> = {
    open: true,
    behavior: result.current.conversationBehavior,
    writerProfile: result.current.writerProfile,
    webResearch: result.current.webResearch,
    projectMemoryMode: result.current.projectMemoryMode,
    fileEditsMode: result.current.fileEditsMode,
    memory: { workspace: { kind: 'available' }, saved: 0, unreadable: [], conflicts: [] },
    isRunning: false,
    onApply: result.current.setConversationSettings,
    onOpenMemoryManager: jest.fn(),
    onOpenMemoryFolder: jest.fn(),
    pinnedContext: pinnedContextPanelPropsStub(),
    memoryExplainerOpen: true,
    onToggleMemoryExplainer: jest.fn(),
    onClose: jest.fn()
  };
  return { host, modes, vscode, result, snapshot, props, message };
}

const rows = [
  { name: 'suggested file edits', key: 'workshop.fileEdits', payload: 'fileEdits', tab: 'Behavior' },
  { name: 'project memory', key: 'workshop.projectMemory', payload: 'projectMemory', tab: 'Memory' }
] as const;

function clickApply() {
  fireEvent.click(screen.getByRole('button', { name: /Analyze/ }));
  fireEvent.click(screen.getByRole('button', { name: 'Apply to next turn' }));
}

function submitted(vscode: ReturnType<typeof createMockVSCode>) {
  return vscode.postMessage.mock.calls.map(([entry]) => entry)
    .filter((entry) => entry.type === MessageType.WORKSHOP_SET_CONVERSATION_SETTINGS).at(-1);
}

describe('PR #236 delayed global-mode snapshots', () => {
  it.each(rows)('an untouched $name row preserves a newer host Off before its snapshot arrives', async (row) => {
    const h = await setup();
    render(<WorkshopConversationBehaviorModal {...h.props} />);
    h.modes[row.key] = 'off'; // The host has changed; React has not heard yet.
    clickApply();
    await h.host.router.route(submitted(h.vscode));
    expect(h.snapshot().payload.session.conversationBehavior.interactionMode).toBe('analysis');
    expect(h.modes[row.key]).toBe('off');
    expect(h.snapshot().payload[row.payload]).toBe('off');
  });

  it.each(rows)('control: an explicit $name pick still writes Ask first over a newer host Off', async (row) => {
    const h = await setup();
    render(<WorkshopConversationBehaviorModal {...h.props} />);
    if (row.tab === 'Memory') fireEvent.click(screen.getByRole('tab', { name: 'Memory' }));
    fireEvent.click(screen.getByRole('button', { name: /^Off/ }));
    fireEvent.click(screen.getByRole('button', { name: /^Ask first/ }));
    if (row.tab === 'Memory') fireEvent.click(screen.getByRole('tab', { name: 'Behavior' }));
    h.modes[row.key] = 'off';
    clickApply();
    await h.host.router.route(submitted(h.vscode));
    expect(h.modes[row.key]).toBe('ask');
    expect(h.snapshot().payload[row.payload]).toBe('ask');
  });

  it('control: the host already preserves both modes when neither field is submitted', async () => {
    const h = await setup();
    h.modes['workshop.fileEdits'] = 'off';
    h.modes['workshop.projectMemory'] = 'off';
    await h.host.router.route(h.message(MessageType.WORKSHOP_SET_CONVERSATION_SETTINGS, {
      behavior: { ...DEFAULT_WORKSHOP_CONVERSATION_BEHAVIOR, interactionMode: 'analysis' },
      writerProfile: DEFAULT_WORKSHOP_WRITER_PROFILE,
      webResearch: DEFAULT_WORKSHOP_WEB_RESEARCH_SETTINGS
    }) as never);
    expect(h.snapshot().payload.session.conversationBehavior.interactionMode).toBe('analysis');
    expect(h.snapshot().payload).toMatchObject({ fileEdits: 'off', projectMemory: 'off' });
  });

  it('the hook must not replace omitted modes with the stale modes it last heard', async () => {
    const h = await setup();
    act(() => h.result.current.setConversationSettings(
      DEFAULT_WORKSHOP_CONVERSATION_BEHAVIOR,
      DEFAULT_WORKSHOP_WRITER_PROFILE,
      DEFAULT_WORKSHOP_WEB_RESEARCH_SETTINGS,
      undefined,
      undefined
    ));
    const payload = submitted(h.vscode).payload;
    expect({ projectMemory: payload.projectMemory, fileEdits: payload.fileEdits })
      .toEqual({ projectMemory: undefined, fileEdits: undefined });
  });
});
