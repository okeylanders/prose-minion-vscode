/** @jest-environment jsdom */

/**
 * The Memory tab and the memory manager through the real WorkshopApp (epic
 * Slice 4b): opening Conversation settings asks for the folder's listing;
 * the tab draws the host's figures; the manager opens over the settings,
 * whose drafts stay; Edit opens Slice 3's sheet; the mode applies with the
 * other drafts. *Alice*, made-up ids.
 */

import * as React from 'react';
import { act, fireEvent, render, screen, within } from '@testing-library/react';
import {
  DEFAULT_WORKSHOP_CONVERSATION_BEHAVIOR,
  DEFAULT_WORKSHOP_WRITER_PROFILE,
  MessageType,
  WorkshopSessionSnapshot,
  WorkshopSessionStateMessage,
  WorkshopTurn
} from '@messages';
import { createMockVSCode } from '@/__tests__/mocks/vscode';

jest.mock('../../../presentation/webview/hooks/useVSCodeApi');
jest.mock('../../../presentation/webview/styles/workshop/tokens.css', () => ({}));
jest.mock('../../../presentation/webview/styles/workshop/shell.css', () => ({}));
jest.mock('../../../presentation/webview/styles/workshop/context.css', () => ({}));
jest.mock('../../../presentation/webview/styles/workshop/session.css', () => ({}));
jest.mock('../../../presentation/webview/styles/workshop/memory.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/widgets/gesturePlayground/gesturePlayground.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/widgets/lexicalGravity/lexicalGravity.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/widgets/creativeVariations/creativeVariations.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/widgets/showVsTell/showVsTell.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/standingDirectiveRail.css', () => ({}));
jest.mock('../../../presentation/webview/components/workshop/schematic/schematic.css', () => ({}));
// eslint-disable-next-line @typescript-eslint/naming-convention -- the mocked module's export is a component name
jest.mock('../../../presentation/webview/components/shared/PmLogo', () => ({ PmLogo: () => null }));

import { useVSCodeApi } from '@hooks/useVSCodeApi';
import { WorkshopApp } from '@/presentation/webview/WorkshopApp';

const session = (overrides: Partial<WorkshopSessionSnapshot> = {}): WorkshopSessionStateMessage => ({
  type: MessageType.WORKSHOP_SESSION_STATE,
  source: 'extension.workshop',
  payload: {
    session: {
      scope: 'open',
      participantSubjectReady: true,
      excerptVersion: 0,
      replacementCount: 0,
      roomRevision: 1,
      contextRevision: 0,
      contextAttachments: [],
      pendingMessageAttachments: [],
      widgetConfigs: [],
      standingDirectives: [],
      todos: [],
      bookmarks: [],
      turns: [],
      turnRewindability: {},
      totalTurns: 0,
      truncatedTurns: 0,
      roomHasMemory: true,
      participants: {
        host: { personaId: 'jill', hasConversation: true },
        toolSidecars: [],
        personaGuests: [],
        chatTarget: { kind: 'host' }
      },
      conversationBehavior: { ...DEFAULT_WORKSHOP_CONVERSATION_BEHAVIOR },
      ...overrides
    },
    writerProfile: { ...DEFAULT_WORKSHOP_WRITER_PROFILE },
    webResearch: { enabled: false },
    projectMemory: 'ask',
    persistence: { available: true, degradedConversationKeys: [] }
  },
  timestamp: 0
});


const deliver = (message: unknown) =>
  act(() => window.dispatchEvent(new MessageEvent('message', { data: message })));

const ENTRY = {
  entryKey: 'a'.repeat(32), id: 'mem-7f3c21', title: 'The Rabbit’s watch runs backward', kind: 'canon', summary: 'From chapter 3 on.',
  status: 'active', pinned: false, created: '2026-11-01T10:00:00Z', updated: '2026-11-03T10:00:00Z', origin: 'writer',
  source: { kind: 'none' }, superseded: false, supersededBy: [], busy: false, fullLoad: null, residency: 'listed'
};
const ESTIMATE = { memories: 1, pinned: 0, pinnedLoaded: 0, pinnedTokens: 0, pinnedOver: false, catalogTokens: 40, frameTokens: 90, pinnedLoadedTokens: 0, catalogEntries: 1 };

describe('PR 222 independent sessions-menu memory checks', () => {
  let vscode: ReturnType<typeof createMockVSCode>;
  const posted = (type: MessageType) => vscode.postMessage.mock.calls
    .map(([message]: [{ type: MessageType; payload: Record<string, unknown> }]) => message)
    .filter((message) => message.type === type);
  const answerListing = () => {
    const requests = posted(MessageType.WORKSHOP_MEMORY_LIST);
    deliver({
      type: MessageType.WORKSHOP_MEMORY_LISTING,
      source: 'extension.workshop',
      payload: { requestId: requests[requests.length - 1].payload.requestId, workspace: { kind: 'available' }, saved: 1, estimate: ESTIMATE, memories: [ENTRY], unreadable: [], conflicts: [] },
      timestamp: 0
    });
  };

  beforeEach(() => {
    jest.useFakeTimers();
    vscode = createMockVSCode();
    (useVSCodeApi as jest.Mock).mockReturnValue(vscode);
    Element.prototype.scrollIntoView = jest.fn();
  });

  afterEach(() => {
    act(() => { jest.runOnlyPendingTimers(); });
    jest.useRealTimers();
    jest.clearAllMocks();
  });


  const flush = () => act(() => { jest.runOnlyPendingTimers(); });
  const openFromMenu = () => {
    fireEvent.click(screen.getByRole('button', { name: /^Sessions/ }));
    fireEvent.click(screen.getByRole('menuitem', { name: 'Memory…' }));
    flush();
    expect(screen.queryByRole('menu')).toBeNull();
    expect(screen.queryByRole('dialog', { name: 'Conversation settings' })).toBeNull();
    return screen.getByRole('dialog', { name: 'Project memories' });
  };
  const listAnswer = (requestId: unknown, title: string, workspace: object = { kind: 'available' }) => deliver({
    type: MessageType.WORKSHOP_MEMORY_LISTING, source: 'extension.workshop', timestamp: 0,
    payload: { requestId, workspace, saved: 1, estimate: ESTIMATE, memories: [{ ...ENTRY, title }], unreadable: [], conflicts: [] }
  });
  const latestList = () => posted(MessageType.WORKSHOP_MEMORY_LIST).at(-1)!.payload.requestId;

  it('opens while memory is Off and the room is running, without changing either; New opens above the manager and Escape closes only the sheet', () => {
    render(<WorkshopApp />);
    const state = session();
    state.payload.projectMemory = 'off';
    state.payload.session.activeRequestId = 'run-review-222';
    deliver(state);
    const manager = openFromMenu();
    answerListing();
    expect(manager.textContent).toContain('Memory is off');
    expect((within(manager).getByRole('button', { name: 'New memory' }) as HTMLButtonElement).disabled).toBe(false);
    fireEvent.click(within(manager).getByRole('button', { name: 'New memory' }));
    flush();
    const sheet = screen.getByRole('dialog', { name: 'Save to project memory' });
    expect(screen.getAllByRole('dialog')).toEqual([manager, sheet]);
    fireEvent.keyDown(window, { key: 'Escape' });
    flush();
    expect(screen.queryByRole('dialog', { name: 'Save to project memory' })).toBeNull();
    expect(screen.getByRole('dialog', { name: 'Project memories' })).toBe(manager);
    expect(posted(MessageType.WORKSHOP_SET_CONVERSATION_SETTINGS)).toHaveLength(0);
    expect(posted(MessageType.WORKSHOP_MEMORY_CREATE)).toHaveLength(0);
  });

  it('drops prior card focus, closes back to the trigger, clears search and ignores an earlier opening’s late answer', () => {
    render(<WorkshopApp />);
    const turn = {
      id: 'turn-4-assistant-1008', role: 'assistant', kind: 'message', participant: 'host', artifact: 'persona_message',
      personaId: 'jill', personaLabel: 'Jill', excerptVersion: 1, content: 'Good.', timestamp: 1,
      memoryProposals: [{ proposes: 'new', title: ENTRY.title, kind: 'canon', summary: 'From chapter 3 on.', body: '', state: 'saved', memoryId: ENTRY.id }]
    } as unknown as WorkshopTurn;
    deliver(session({ turns: [turn], totalTurns: 1 }));
    fireEvent.click(screen.getByRole('button', { name: 'Open in manager' }));
    flush();
    answerListing();
    expect(screen.getByRole('dialog', { name: 'Project memories' }).querySelector('[aria-current="true"]')).not.toBeNull();
    fireEvent.click(screen.getByRole('button', { name: 'Close project memories' }));
    flush();
    const manager = openFromMenu();
    const oldRequest = latestList();
    answerListing();
    expect(manager.querySelector('[aria-current="true"]')).toBeNull();
    const search = within(manager).getByRole('searchbox', { name: 'Search project memories' });
    fireEvent.change(search, { target: { value: 'watch' } });
    expect(posted(MessageType.WORKSHOP_MEMORY_LIST).at(-1)!.payload.query).toBe('watch');
    fireEvent.keyDown(window, { key: 'Escape' });
    flush();
    expect(screen.queryByRole('dialog', { name: 'Project memories' })).toBeNull();
    expect(document.activeElement).toBe(screen.getByRole('button', { name: /^Sessions/ }));
    const reopened = openFromMenu();
    expect((within(reopened).getByRole('searchbox', { name: 'Search project memories' }) as HTMLInputElement).value).toBe('');
    listAnswer(latestList(), 'Newest listing');
    listAnswer(oldRequest, 'Obsolete listing');
    expect(reopened.textContent).toContain('Newest listing');
    expect(reopened.textContent).not.toContain('Obsolete listing');
    fireEvent.mouseDown(reopened.parentElement!);
    flush();
    expect(screen.queryByRole('dialog', { name: 'Project memories' })).toBeNull();
    expect(document.activeElement).toBe(screen.getByRole('button', { name: /^Sessions/ }));
  });

  it.each([
    [{ kind: 'unavailable', reason: 'no-workspace' }, 'no folder is open'],
    [{ kind: 'unavailable', reason: 'multi-root' }, 'several folders'],
    [{ kind: 'unknown' }, 'couldn’t be listed']
  ])('keeps the new door available but shows the host’s unavailable or failure answer: %j', (workspace, words) => {
    render(<WorkshopApp />);
    deliver(session());
    const manager = openFromMenu();
    expect(manager.textContent).toContain('Reading this project’s memories');
    listAnswer(latestList(), ENTRY.title, workspace as object);
    expect(manager.textContent).toContain(words as string);
    expect((within(manager).getByRole('button', { name: 'New memory' }) as HTMLButtonElement).disabled).toBe(true);
    expect((within(manager).getByRole('button', { name: /^Archive /  }) as HTMLButtonElement).disabled).toBe(true);
    expect((within(manager).getByRole('searchbox', { name: 'Search project memories' }) as HTMLInputElement).disabled).toBe(true);
  });

  it('manager Archive from the new door uses the existing route and shows a correlated host refusal', () => {
    render(<WorkshopApp />);
    deliver(session());
    const manager = openFromMenu();
    answerListing();
    fireEvent.click(within(manager).getByRole('button', { name: /^Archive /  }));
    const request = posted(MessageType.WORKSHOP_MEMORY_FORGET).at(-1)!.payload;
    expect(request).toMatchObject({ memoryId: ENTRY.id, action: 'forget' });
    expect((within(manager).getByRole('button', { name: /^Archive /  }) as HTMLButtonElement).disabled).toBe(true);
    deliver({ type: MessageType.WORKSHOP_MEMORY_WRITE_RESULT, source: 'extension.workshop', timestamp: 0,
      payload: { requestId: request.requestId, ok: false, error: 'Another window is editing that memory. Try again in a moment.' } });
    expect(within(manager).getByRole('alert').textContent).toContain('Another window is editing');
    answerListing();
    expect((within(manager).getByRole('button', { name: /^Archive /  }) as HTMLButtonElement).disabled).toBe(false);
  });
});
